المعرفات المرجعية يجب أن تكون UUID أو TEXT. لا تستخدم bigint إذا كانت branches.id أو couriers.id من نوع UUID.

-- إذا كانت IDs في الجداول الأصلية UUID استخدم UUID، أو استخدم TEXT كحل مرن.
alter table public.orders alter column branch_id type text using branch_id::text;
alter table public.orders alter column courier_id type text using courier_id::text;
alter table public.couriers alter column branch_id type text using branch_id::text;
alter table public.settlements alter column branch_id type text using branch_id::text;
alter table public.settlements alter column courier_id type text using courier_id::text;

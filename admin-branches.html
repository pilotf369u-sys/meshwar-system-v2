مخطط Supabase المطلوب لهذه المرحلة

إن ظهرت رسالة أن جدول/عمود غير موجود، نفّذ هذا SQL مرة واحدة من Supabase SQL Editor. مفتاح Publishable لا يستطيع إنشاء جداول PostgreSQL.

alter table public.orders add column if not exists branch_id bigint;
alter table public.orders add column if not exists branch_name text;
alter table public.orders add column if not exists courier_id bigint;
alter table public.orders add column if not exists courier_name text;
alter table public.orders add column if not exists delivery_fee numeric default 0;
alter table public.orders add column if not exists settlement_id bigint;

create table if not exists public.couriers (
  id bigint generated by default as identity primary key,
  branch_id bigint not null,
  name text not null,
  phone text not null,
  password text not null,
  delivery_fee numeric default 0,
  is_active boolean default true,
  created_at timestamptz default now()
);

create table if not exists public.settlements (
  id bigint generated by default as identity primary key,
  settlement_type text not null default 'courier',
  branch_id bigint not null,
  courier_id bigint,
  gross_amount numeric default 0,
  delivery_fee numeric default 0,
  net_amount numeric default 0,
  currency text default '$',
  order_ids jsonb default '[]'::jsonb,
  status text not null default 'pending_admin',
  requested_by text,
  requested_at timestamptz default now(),
  approved_by bigint,
  approved_at timestamptz
);
alter table public.settlements add column if not exists currency text default '$';
create index if not exists idx_orders_branch_id on public.orders(branch_id);
create index if not exists idx_orders_courier_id on public.orders(courier_id);
create index if not exists idx_couriers_branch_id on public.couriers(branch_id);
create index if not exists idx_settlements_branch_id on public.settlements(branch_id);
